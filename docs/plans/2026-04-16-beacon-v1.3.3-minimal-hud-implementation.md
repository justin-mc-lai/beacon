# beacon/utils/minimal_hud.py
"""Minimal HUD for Beacon v1.3.3 - displays version, workflow, and execution status."""

from __future__ import annotations

from typing import Any, Literal, TypedDict


class HUDState(TypedDict, total=False):
    beacon_version: str
    docs_version: str
    feature: str
    stage: Literal["requirement", "implement", "qa", "release"]
    requirement_complete: bool
    execution: dict[str, Any] | None


def create_hud_state(
    beacon_version: str,
    docs_version: str,
    feature: str,
    stage: Literal["requirement", "implement", "qa", "release"],
    requirement_complete: bool = False,
    execution: dict[str, Any] | None = None,
) -> HUDState:
    """Create HUD state dictionary."""
    return {
        "beacon_version": beacon_version,
        "docs_version": docs_version,
        "feature": feature,
        "stage": stage,
        "requirement_complete": requirement_complete,
        "execution": execution,
    }
```

# beacon/utils/minimal_hud.py (add to existing file)

def render_context_line(state: HUDState) -> str:
    """Render the first line (context layer) of HUD."""
    version_info = f"[beacon:{state['beacon_version']}] docs:{state['docs_version']}"
    feature = state["feature"][:30]  # Truncate to 30 chars
    
    # Workflow stages
    stages = []
    
    # Requirement stage
    if state["stage"] == "requirement":
        stages.append("[需求]")
    elif state.get("requirement_complete"):
        stages.append("需求✓")
    else:
        stages.append("需求")
    
    # Implement stage
    if state["stage"] == "implement":
        stages.append("[执行]")
    else:
        stages.append("执行")
    
    # QA and Release stages
    stages.extend(["QA", "发布"])
    
    workflow = " ".join(stages)
    
    return f"{version_info} | {feature} | {workflow}"
```

# beacon/utils/minimal_hud.py (add to existing file)

def render_execution_line(state: HUDState) -> str:
    """Render the second line (execution layer) of HUD."""
    execution = state.get("execution")
    if not execution:
        return ""
    
    mode = execution["mode"]
    
    if mode == "single":
        return _render_single_mode(execution)
    elif mode == "team":
        return _render_team_mode(execution)
    elif mode == "ralph":
        return _render_ralph_mode(execution)
    
    return ""


def _render_single_mode(execution: dict[str, Any]) -> str:
    """Render single mode execution line."""
    parts = ["▸ 执行中"]
    
    # Completed files
    for file in execution.get("completed_files", []):
        parts.append(f"{file}✓")
    
    # Current file
    if current := execution.get("current_file"):
        parts.append(f"{current}◐")
    
    # Elapsed time
    if elapsed := execution.get("elapsed"):
        parts.append(f"⏱ {elapsed}")
    
    return " | ".join(parts)
```

def _render_ralph_mode(execution: dict[str, Any]) -> str:
    """Render ralph mode execution line."""
    parts = []
    
    # Round info
    round_num = execution.get("round", 1)
    parts.append(f"Round {round_num}/∞")
    
    # Circuit breaker status
    circuit = execution.get("circuit", {})
    circuit_status = circuit.get("status", "ok")
    parts.append(f"circuit:{circuit_status}")
    
    # Files
    file_parts = []
    for file in execution.get("completed_files", []):
        file_parts.append(f"{file}✓")
    if current := execution.get("current_file"):
        file_parts.append(f"{current}◐")
    
    if file_parts:
        parts.append(" ".join(file_parts))
    
    # Elapsed
    if elapsed := execution.get("elapsed"):
        parts.append(f"⏱ {elapsed}")
    
    return "▸ " + " | ".join(parts)
```

def render_hud(state: HUDState) -> list[str]:
    """Render complete HUD, returns 1-3 lines."""
    lines = []
    
    # Line 1: Always show context
    lines.append(render_context_line(state))
    
    # Line 2: Show execution if in implement stage
    if state["stage"] == "implement" and state.get("execution"):
        exec_line = render_execution_line(state)
        if exec_line:
            lines.append(exec_line)
    
    return lines
```

import json
from pathlib import Path

def save_hud_state(state: HUDState, path: Path) -> None:
    """Save HUD state to JSON file."""
    path.parent.mkdir(parents=True, exist_ok=True)
    with open(path, "w") as f:
        json.dump(state, f, indent=2)

def load_hud_state(path: Path) -> HUDState | None:
    """Load HUD state from JSON file."""
    if not path.exists():
        return None
    with open(path) as f:
        return json.load(f)
```
